-- 1.Display all records.
select * from store_sales3;

-- Show customer name, city and product purchased.
select customer_name, city, product_name from store_sales3;

-- Find all customers from Delhi.
select customer_name from store_sales3 where city = 'Delhi';

-- 	Display all Electronics products.
select product_name from store_sales3 where category = 'Electronics';

-- Find sales greater than ₹30,000.
SELECT * FROM store_sales3 WHERE total_amount > 30000;

--  Show customers aged above 30 years
select * from store_sales3 WHERE age >=30; 

-- Sort records by highest sale amount.
SELECT * FROM store_sales3 ORDER BY total_amount DESC;

-- Display the first 5 records.
SELECT * FROM store_sales3 LIMIT 5;

-- Find customers whose names start with 'R'.
SELECT customer_name FROM store_sales3 WHERE customer_name LIKE 'R%';

-- 	Display all unique cities.
SELECT DISTINCT city FROM store_sales3;

-- Which city generates the highest sales?
SELECT city, SUM(total_amount) AS total_sales FROM store_sales3 GROUP BY city
ORDER BY total_sales DESC LIMIT 1;

-- Which product category earns the maximum revenue?
SELECT category, SUM(total_amount) AS total_revenue FROM store_sales3 GROUP BY category
ORDER BY total_revenue DESC LIMIT 1;

-- Which salesperson performs the best?
SELECT salesperson, SUM(total_amount) AS total_sales FROM store_sales3 GROUP BY salesperson
ORDER BY total_sales DESC LIMIT 1;

-- Which payment mode is most preferred?
SELECT payment_mode, COUNT(*) AS total_orders FROM store_sales3 GROUP BY payment_mode
ORDER BY total_orders DESC LIMIT 1;

-- Which customers purchase frequently?
SELECT customer_name, COUNT(*) AS purchase_count FROM store_sales3 GROUP BY customer_name
ORDER BY purchase_count DESC;

-- What are the monthly sales trends?
SELECT MONTH(order_date) AS month,SUM(total_amount) AS total_sales
FROM store_sales3
GROUP BY MONTH(order_date)
ORDER BY month;
